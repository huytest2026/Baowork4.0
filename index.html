function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Đọc dữ liệu đã lưu từ sheet "Số hóa tuyến cáp" var sheetData = ss.getSheetByName("Số hóa tuyến cáp"); var rowsData = sheetData ? sheetData.getDataRange().getValues() : []; if (rowsData.length > 0) rowsData.shift(); // Bỏ dòng tiêu đề var data = rowsData.map(function(row) { return { id: row[0], tramQuanLy: row[1], tenTuyenQuyDinh: row[2], link: row[3], tuyenCap: row[4], chieuDaiLink: row[5], doiTuong: row[6], phuongThucThiCong: row[7], otdrMx: row[8], doSau: row[9], layToaDo: row[10], thoiGian: row[11], chieuTu: row[12], dienGiai: row[13], km: row[14], m: row[15], chupAnh: row[16], taiKm: row[17], tuyenDuong: row[18], cachMepDuong: row[19], lyDoCapNhat: row[20], soMxPhatSinh: row[21], soCapPhatSinh: row[22], coTraoSoi: row[23], otdrMxCap: row[24] }; }); // 2. Đọc dữ liệu thư viện từ sheet "LIBRARY" var sheetLib = ss.getSheetByName("LIBRARY"); var rowsLib = sheetLib ? sheetLib.getDataRange().getValues() : []; if (rowsLib.length > 0) rowsLib.shift(); var library = rowsLib.map(function(row) { return { tram: row[1], tuyen: row[2], link: row[3], tenTuyen: row[5], tuyenDuong: row[7], chieuDai: row[8] }; }); var result = { data: data, library: library }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Số hóa tuyến cáp"); var data = JSON.parse(e.postData.contents); var fileUrl = data.chupAnh || ""; if (data.imageBytes && data.imageName) { var folder; var folders = DriveApp.getFoldersByName("QuanLyCapQuang_Images"); if (folders.hasNext()) { folder = folders.next(); } else { folder = DriveApp.createFolder("QuanLyCapQuang_Images"); } var base64Data = data.imageBytes.split(',')[1] || data.imageBytes; var decodedBytes = Utilities.base64Decode(base64Data); var blob = Utilities.newBlob(decodedBytes, data.imageType || "image/jpeg", data.imageName); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); fileUrl = file.getUrl(); } // TÍNH TOÁN OTDR ->MX _cáp VỚI LOGIC SẮP XẾP TỪ BẮC VÀO NAM & ĐỒNG BỘ KM var otdrMxCapVal = ""; if (data.lyDoCapNhat === 'Sự cố cáp quang' && data.otdrMx !== "" && data.otdrMx !== null && data.otdrMx !== undefined) { var currentOtdr = parseFloat(data.otdrMx) || 0; var tram = data.tramQuanLy; var tuyen = data.tuyenCap; var currentId = data.id; var rows = sheet.getDataRange().getValues(); var routeRows = []; for (var i = 1; i < rows.length; i++) { var rLat = 0; var rLng = 0; var coords = String(rows[i][10] || "").split(","); if (coords.length === 2) { rLat = parseFloat(coords[0]) || 0; rLng = parseFloat(coords[1]) || 0; } routeRows.push({ id: rows[i][0], tramQuanLy: rows[i][1], tuyenCap: rows[i][4], otdrMx: parseFloat(rows[i][8]) || 0, soCapPhatSinh: parseFloat(rows[i][22]) || 0, lat: rLat, // Vĩ độ phục vụ sắp xếp Bắc -> Nam km: parseFloat(rows[i][14]) || 0, m: parseFloat(rows[i][15]) || 0 }); } // Lọc đúng Trạm và Tuyến cáp var filteredRoute = routeRows.filter(function(r) { return r.tramQuanLy === tram && r.tuyenCap === tuyen; }); // SẮP XẾP THỨ TỰ TỪ BẮC VÀO NAM (Vĩ độ giảm dần từ Bắc xuống Nam, kết hợp đồng bộ theo Km/M và OTDR) filteredRoute.sort(function(a, b) { if (a.lat !== 0 && b.lat !== 0 && Math.abs(a.lat - b.lat) > 0.0001) { return b.lat - a.lat; // Vĩ độ lớn hơn nằm ở phía Bắc hơn } if (a.km !== b.km) { return a.km - b.km; } if (a.m !== b.m) { return a.m - b.m; } return a.otdrMx - b.otdrMx; }); // Tính tổng m cáp phát sinh phía trước điểm sự cố theo chuẩn Bắc -> Nam var sumMcp = 0; for (var j = 0; j < filteredRoute.length; j++) { if (filteredRoute[j].otdrMx < currentOtdr) { if (!currentId || String(filteredRoute[j].id) !== String(currentId)) { sumMcp += filteredRoute[j].soCapPhatSinh; } } } otdrMxCapVal = Number((currentOtdr + (sumMcp * 1.467700 / 1000)).toFixed(4)); } var rowDataValues = [ "", data.tramQuanLy || "", data.tenTuyenQuyDinh || "", data.link || "", data.tuyenCap || "", data.chieuDaiLink || "", data.doiTuong || "", data.phuongThucThiCong || "", data.otdrMx || "", data.doSau || "", data.layToaDo || "", data.thoiGian || "", data.chieuTu || "", data.dienGiai || "", data.km || "", data.m || "", fileUrl, data.taiKm || "", data.tuyenDuong || "", data.cachMepDuong || "", data.lyDoCapNhat || "", data.soMxPhatSinh || "", data.soCapPhatSinh || "", data.coTraoSoi || "", otdrMxCapVal ]; var targetRow = -1; if (data.id !== undefined && data.id !== null && data.id !== "") { var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (rows[i][0] == data.id) { targetRow = i + 1; break; } } } if (targetRow > 0) { rowDataValues[0] = data.id; sheet.getRange(targetRow, 1, 1, rowDataValues.length).setValues([rowDataValues]); } else { var lastRow = sheet.getLastRow(); var nextId = 1; if (lastRow > 1) { var ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues(); var maxId = 0; for (var j = 0; j < ids.length; j++) { if (Number(ids[j][0]) > maxId) { maxId = Number(ids[j][0]); } } nextId = maxId + 1; } rowDataValues[0] = nextId; sheet.appendRow(rowDataValues); } return ContentService.createTextOutput(JSON.stringify({"result": "success", "id": rowDataValues[0], "url": fileUrl})) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": error.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
